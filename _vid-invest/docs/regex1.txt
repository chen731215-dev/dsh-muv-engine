     * 只认 地图/世界/地点 格式，一个标签只生成一张图
     * 处理两种情况：
     *   A. <img src="地图/公司本部/公司天台"> — src 属性中包含 地图/
     *   B. 文本节点中包含 地图/公司本部/公司天台 — <img>地图/... 被浏览器解析为 void 元素后
     * 匹配不到时回退到同世界随机地点 */
    function processMapTags(){
      try{
        const content=document.getElementById('readingContent');
        const worldKeys=Object.keys(WORLD_MAP_PLACES).join('|');
        const mapRe=new RegExp('地图\\/('+worldKeys+')\\/([^<&\\s,。！？、；：\\/]+)');
        /* 兼容省略世界名的写法：地图/公司1 → 全地图模糊匹配 */
        const mapReLoose=/地图\/([^<&\s,。！？、；：\/]+)/;
        const _matchTag=(text)=>{
          const mf=text.match(mapRe), ml=text.match(mapReLoose);
          if(mf&&(!ml||mf.index<=ml.index)) return {world:mf[1],keyword:mf[2],index:mf.index,len:mf[0].length};
          if(ml) return {world:null,keyword:ml[1],index:ml.index,len:ml[0].length};
          return null;
        };
        const _resolveAny=(kw)=>{
          const worlds=Object.keys(WORLD_MAP_PLACES), hits=[];
          for(const w of worlds){
            const r=_findBestMatchPlace(w,kw);
            if(r&&!r.fallback) hits.push({world:w,res:r});
          }
          if(hits.length){ const p=hits[Math.floor(Math.random()*hits.length)]; return {world:p.world,res:p.res}; }
          const w=worlds[Math.floor(Math.random()*worlds.length)];
          return {world:w,res:_findBestMatchPlace(w,kw)};
        };
        const _makeContainer=(worldName,bestPlace,tipText)=>{
          const imgSrc=`${BASE_URL}插图/地图/${bestPlace}.webp`;
          const id='map-'+Math.random().toString(36).substr(2,9);
          const mapLabel=`${worldName} · ${bestPlace}`;
          const subTip=tipText?`<div class="image-sub-tip">${tipText}</div>`:'';
          const div=document.createElement('div');
          div.className='image-container map-image-container';
          div.setAttribute('data-map-world',worldName);
          div.setAttribute('data-map-place',bestPlace);
          div.innerHTML=
            `<div class="image-controls">`+
            `<button class="image-toggle-btn" onclick="toggleImage('${id}')">${mapLabel}</button>`+
            `</div>${subTip}`+
            `<div class="image-content expanded" id="${id}">`+
            `<img src="${imgSrc}" alt="${mapLabel}" loading="lazy" onerror="this.onerror=null;this.alt='图片加载失败';this.style.opacity='0.3';">`+
            `</div>`;
          return div;
        };
        const _resolve=(w,p)=>{
          if(!WORLD_MAP_PLACES[w])return null;
          return _findBestMatchPlace(w,p);
        };
        const _build=(w,pr)=>{
          if(!w){
            const any=_resolveAny(String(pr).trim());
            if(!any||!any.res) return null;
            return _makeContainer(any.world,any.res.place,
              `无地点「${any.res.origKeyword}」插图，已替换为「${any.world} · ${any.res.place}」`);
          }
          const r=_resolve(String(w).trim(),String(pr).trim());
          if(!r)return null;
          /* 精确匹配不显示提示；模糊匹配和回退都显示 */
          const placeBase=r.place.replace(/\d+$/,'');
          const isExact=placeBase===r.origKeyword||placeBase.replace(/[·\s]/g,'')===r.origKeyword.replace(/[·\s]/g,'');
          const tip=isExact?null:`无地点「${r.origKeyword}」插图，已替换为「${r.place}」`;
          return _makeContainer(String(w).trim(),r.place,tip);
        };

        /* A. 处理 <img src="地图/世界/地点"> — src 属性中包含 地图/ */
        content.querySelectorAll('img').forEach(img=>{
          const src=img.getAttribute('src')||'';
          const m=_matchTag(src);
          if(m){
            const c=_build(m.world,m.keyword);
            if(c) img.replaceWith(c);
          }
        });

        /* B. 处理文本节点中的 地图/世界/地点 — TreeWalker 遍历 */
        const walker=document.createTreeWalker(content,NodeFilter.SHOW_TEXT,null);
        const targets=[];
        let node;
        while((node=walker.nextNode())){
          const text=node.nodeValue;
          if(!text||!text.includes('地图/')) continue;
          const m=_matchTag(text);
          if(m) targets.push({node,match:m,text});
        }
        for(const t of targets){
          const c=_build(t.match.world,t.match.keyword);
          if(c){
            const before=t.text.substring(0,t.match.index);
            const after=t.text.substring(t.match.index+t.match.len);
            const parent=t.node.parentNode;
            if(before) parent.insertBefore(document.createTextNode(before),t.node);
            parent.insertBefore(c,t.node);
            if(after) parent.insertBefore(document.createTextNode(after),t.node);
            parent.removeChild(t.node);
          }
        }

        /* 清理：删除无 src 的空 <img> 元素 */
        content.querySelectorAll('img').forEach(img=>{
          if(!img.getAttribute('src')) img.remove();
        });
      }catch(e){ /* 地图处理出错时不影响 CG 流程 */ }
    }

    function processDialogue() {
      const content = document.getElementById('readingContent');
      const paras = content.innerHTML.trim().split(/\n\s*\n+/).filter(p=>p.trim());
      content.innerHTML = paras.map(p => {
        p = p.trim();

        const re = /([^\s"'"'「」]+)[\s"'"'「」]*[「"'](.+?)[」"']/g;
        let out = '';
        let last = 0;
        let m;
        let matched = false;
        while ((m = re.exec(p)) !== null) {
          const charName = normalizeCharacterName(m[1]);
          if (!charName) continue; // 名字不在配置中，跳过该匹配
          matched = true;
          const lead = p.slice(last, m.index).trim();
          if (lead) out += `<p>${lead}</p>`;
          const c = CHARACTER_CONFIG[charName];
          const fn = charName.charAt(0);
          const fr = charName.slice(1);
          out += `<div class="dialogue-container">
              <div class="char-avatar-wrap" style="border:2.5px solid ${c.border};">
                <img src="${c.avatar}" alt="${charName}"
                  onerror="this.style.display='none';">
              </div>
              <div class="bubble-wrap">
                <div class="character-name" style="color:${c.color};">
                  <span class="cn-first">${fn}</span><span class="cn-rest">${fr}</span>
                </div>
                <div class="bubble-text">${m[2]}<span class="bubble-pendant">${pendantFor(charName)}</span></div>
              </div>
            </div>`;
          last = m.index + m[0].length;
        }
        if (matched) {
          const tail = p.slice(last).trim();
          if (tail) out += `<p>${tail}</p>`;
          return out;
        }
        return `<p>${p}</p>`;

      let d=IMAGE_LISTS[char]; if(d?.[type]) return d[type];
      const nc=char.trim().normalize('NFC');
      if(!d){ const k=Object.keys(IMAGE_LISTS).find(c=>c.trim().normalize('NFC')===nc||c.replace(/\s/g,'')===char.replace(/\s/g,'')); if(k) d=IMAGE_LISTS[k]; }
      if(!d) return 0;
      if(d[type]) return d[type];
      const nt=type.trim().normalize('NFC');
      const k2=Object.keys(d).find(t=>t.trim().normalize('NFC')===nt||t.replace(/\s/g,'')===type.replace(/\s/g,''));
      return k2?d[k2]:0;
    }
    function _pickSubChar(originalChar){
      const pool=_ALL_CHAR_NAMES.filter(c=>c!==originalChar);
      return pool[Math.floor(Math.random()*pool.length)]||_ALL_CHAR_NAMES[0];
    }
    function _pickSubType(cat,excludeType){
      const isSfw=safeDecode(cat)==='SFW';
      const src=isSfw?_SFW_COMMON:_NSFW_COMMON;
      const keys=Object.keys(src).filter(k=>k!==excludeType);
      if(!keys.length) return Object.keys(src)[0]||excludeType;
      return keys[Math.floor(Math.random()*keys.length)];
    }
    function _charExists(char){
      if(IMAGE_LISTS[char]) return true;
      const nc=char.trim().normalize('NFC');
      return Object.keys(IMAGE_LISTS).some(c=>c.trim().normalize('NFC')===nc||c.replace(/\s/g,'')===char.replace(/\s/g,''));
    }
    function _sceneExists(cat,scene){
      const src=cat==='SFW'?_SFW_COMMON:_NSFW_COMMON;
      if(src[scene]) return true;
      const ns=scene.trim().normalize('NFC');
      return Object.keys(src).some(k=>k.trim().normalize('NFC')===ns||k.replace(/\s/g,'')===scene.replace(/\s/g,''));
    }
    function autoAssignImageNumber(src) {
      const parts=src.split('/');
      const idx=_findCgIndex(parts);
      if(idx<0||idx+2>=parts.length) return {src,tip:null};
      const cat=parts[idx]; let char=safeDecode(parts[idx+1]);
      let name=safeDecode(parts[idx+2]).replace(/\.webp$/,'');
      let tip=null;

      /* 剥离数字后缀，提取基础场景名 */
      const fm=name.match(/^(.+?)(\d+)$/);
      const baseName=fm?fm[1]:name;
      const hasNum=!!fm;
      const charOk=_charExists(char);
      const sceneOk=charOk ? !!_cgLookup(char,baseName) : _sceneExists(safeDecode(cat),baseName);

      /* 正常：角色+场景都存在 */
      if(charOk && sceneOk){
        if(hasNum) return {src,tip:null};
        const max=_cgLookup(char,baseName);
        parts[idx+2]=encodeURIComponent(baseName+(Math.floor(Math.random()*max)+1)+'.webp');
        return {src:parts.join('/'),tip:null};
      }

      /* 情况1：角色不存在 + 场景存在 → 只换角色，保留场景 */
      if(!charOk && sceneOk){
        const origChar=char;
        const subChar=_pickSubChar(char);
        parts[idx+1]=encodeURIComponent(subChar);
        const max=_cgLookup(subChar,baseName);
        if(!max) return {src:parts.join('/'),tip:null};
        parts[idx+2]=encodeURIComponent(baseName+(Math.floor(Math.random()*max)+1)+'.webp');
        tip=`角色「${origChar}」无插图，已替换为「${subChar} · ${baseName}」`;
        return {src:parts.join('/'),tip};
      }

      /* 情况2：角色不存在 + 场景也不存在 → 换角色+换场景 */
      if(!charOk && !sceneOk){
        const origChar=char, origType=baseName;
        const subChar=_pickSubChar(char);
        const subType=_pickSubType(cat,baseName);
        parts[idx+1]=encodeURIComponent(subChar);
        const max=_cgLookup(subChar,subType);
        if(!max) return {src:parts.join('/'),tip:`角色「${origChar}」场景「${origType}」无插图`};
        parts[idx+2]=encodeURIComponent(subType+(Math.floor(Math.random()*max)+1)+'.webp');
        tip=`「${origChar}」无插图，「${origType}」不存在，已替换为「${subChar} · ${subType}」`;
        return {src:parts.join('/'),tip};
      }

      /* 情况3：角色存在 + 场景不存在 → 只换场景 */
      const origType=baseName;
      const subType=_pickSubType(cat,baseName);
      const max=_cgLookup(char,subType);
      if(!max) return {src:parts.join('/'),tip:null};
      parts[idx+2]=encodeURIComponent(subType+(Math.floor(Math.random()*max)+1)+'.webp');
      tip=`「${char}」无「${origType}」插图，已替换为「${subType}」`;
      return {src:parts.join('/'),tip};
    }

    function createInteractiveImageContainer(imgHTML) {
      const id = 'img-' + Math.random().toString(36).substr(2,9);
      const m = imgHTML.match(/src=["']([^"']+)["']/i);
      let tipText=null;
      if (m) {
        const result=autoAssignImageNumber(m[1]);
        imgHTML = imgHTML.replace(m[1], result.src);
        tipText=result.tip;
      }
      const subTip = tipText
        ? `<div class="image-sub-tip">${tipText}</div>`
        : '';
      return `<div class="image-container">
        <div class="image-controls">
          <button class="image-toggle-btn" onclick="toggleImage('${id}')">查看插画</button>
          <button class="image-refresh-btn" onclick="refreshImage('${id}',event)" title="随机切换">↻</button>
        </div>
        ${subTip}
        <div class="image-content expanded" id="${id}">${imgHTML}</div>
      </div>`;
    }

    window.toggleImage = id => {
      const d = document.getElementById(id);
      const btn = d.parentElement.querySelector('.image-toggle-btn');
      const expanded = d.classList.toggle('expanded');
      const mapContainer = d.parentElement.closest('.map-image-container');
      if (mapContainer) {
        const label = `${mapContainer.dataset.mapWorld||''} · ${mapContainer.dataset.mapPlace||''}`.trim();
        btn.textContent = expanded ? label : `查看：${label}`;
      } else {
        btn.textContent = expanded ? '收起' : '查看插画';
      }
    };
    window.refreshImage = (id,e) => {
      const img = document.getElementById(id).querySelector('img');
      if(!img) return;
      const btn = e?.target?.closest('.image-refresh-btn');
      if (btn) { btn.style.transform='rotate(360deg)'; setTimeout(()=>btn.style.transform='',500); }
      img.style.opacity = '0.4';
      img.style.transition = 'opacity 0.3s ease';
      const _updateTip=(container,text)=>{
        if(!container) return;
        let tip=container.querySelector('.image-sub-tip');
        if(!tip){ tip=document.createElement('div'); tip.className='image-sub-tip'; container.querySelector('.image-content').before(tip); }
        tip.textContent=text;
      };
      try{
        const parts = img.src.split('/');
        const idx = _findCgIndex(parts);
        if(idx<0||idx+2>=parts.length){ img.style.opacity='1'; return; }
        const cat = parts[idx]; let char = safeDecode(parts[idx+1]);
        const rawFile = safeDecode(parts[idx+2]).replace(/\.webp$/,'');
        const fm = rawFile.match(/^(.+?)(\d+)$/);
        let type = fm ? fm[1] : rawFile;
        const curNum = fm ? parseInt(fm[2]) : 0;
        const container=img.closest('.image-container');

        /* 情况1：角色不存在 → 替换角色 */
        if(!_charExists(char)){
          const origChar=char;
          const subChar=_pickSubChar(char);
          parts[idx+1]=encodeURIComponent(subChar);
          char=subChar;
          let max=_cgLookup(char,type);
          if(!max){
            const origType=type;
            const subType=_pickSubType(cat,type);
            max=_cgLookup(char,subType)||1;
            type=subType;
            _updateTip(container,`「${origChar}」无插图，「${origType}」不存在，已替换为「${subChar} · ${subType}」`);
          }else{
            _updateTip(container,`角色「${origChar}」无插图，已替换为「${subChar} · ${type}」`);
          }
          let newNum; if(max===1) newNum=1; else do{ newNum=Math.floor(Math.random()*max)+1; }while(newNum===curNum);
          parts[idx+2]=encodeURIComponent(type+newNum+'.webp');
          img.onload=()=>{img.style.opacity='1';}; img.onerror=()=>{img.style.opacity='1';};
          img.src=parts.join('/');
          return;
        }

        /* 情况2：角色存在但场景不存在 → 替换场景 */
        let max = _cgLookup(char,type);
        if(!max){
          const origType=type;
          const subType=_pickSubType(cat,type);
          max=_cgLookup(char,subType)||1;
          type=subType;
          _updateTip(container,`「${char}」无「${origType}」插图，已替换为「${subType}」`);
        }

        let newNum;
        if(max===1) newNum=1;
        else do{ newNum=Math.floor(Math.random()*max)+1; }while(newNum===curNum);
        parts[idx+2] = encodeURIComponent(type+newNum+'.webp');
        img.onload = ()=>{ img.style.opacity='1'; };
        img.onerror = ()=>{ img.style.opacity='1'; };
        img.src = parts.join('/');
      }catch(err){ img.style.opacity='1'; }
    };

    function processAllImages() {
      document.getElementById('readingContent').querySelectorAll('img').forEach(img => {
        if (img.closest('.image-container') || img.closest('.char-avatar-wrap')) return;
        /* 跳过没有 src 的空 <img>（地图标签残留 void 元素） */
        if (!img.getAttribute('src')) { img.remove(); return; }
        /* 跳过 src 中包含 地图/ 的图片（地图图片不归 CG 管） */
        if (img.getAttribute('src').includes('地图/')) return;
        const wrap = document.createElement('div');
        wrap.innerHTML = createInteractiveImageContainer(img.outerHTML);
        const el = wrap.firstElementChild;
        img.replaceWith(el);
        /* 图床尚未上传该插图时：显示占位提示，不显示破图 */
        const newImg = el.querySelector('img');
        if (newImg) {
          newImg.addEventListener('error', () => {
            newImg.style.display = 'none';
            const box = newImg.parentElement;
            if (box && !box.querySelector('.img-miss')) {
              const d = document.createElement('div');
              d.className = 'img-miss';
              d.textContent = '插图暂未上传 · 图床同步后自动显示';
              box.appendChild(d);
            }
          }, { once: true });
        }
      });
    }

    /* ════════ 音频：<audio>日常1</audio> -> 自动播放一次（不循环） ════════
     * 识别正则：/<audio>(.*?)<\/audio>/g
     * 替换为：<audio id="audio-element" preload="auto" autoplay>
     *           <source src="https://zyxjack123.top/足控天堂/音频/$1.mp3" type="audio/mpeg">
     *         </audio>
     */
    function processAudio() {
      const content = document.getElementById('readingContent');
      if (!content) return;
      const AUDIO_RE = /<audio>(.*?)<\/audio>/g;
      if (!AUDIO_RE.test(content.innerHTML)) return;
      let lastName = '';
      content.innerHTML = content.innerHTML.replace(/<audio>(.*?)<\/audio>/g, (m, name) => {
        const n = String(name).trim();
        lastName = n;
        return `<span class="audio-tip">♪ 场景音乐 · 点击左上角 LOGO 可暂停 / 继续</span>` +
          `<audio id="audio-element" preload="auto" autoplay>` +
          `<source src="${BASE_URL}音频/${n}.mp3" type="audio/mpeg"></audio>`;
      });
      const audio = document.getElementById('audio-element');
      if (!audio) return;
      /* 只播放一次：不设 loop，播完即停 */
      audio.loop = false;
      audio.addEventListener('play',  syncAudioIcon);
      audio.addEventListener('pause', syncAudioIcon);
      audio.addEventListener('ended', syncAudioIcon);
      audio.addEventListener('error', () => {
        const tip = audio.closest('p');
        if (tip) tip.style.opacity = '.55';
      });
      const p = audio.play();
      if (p && p.catch) p.catch(() => {});
      syncAudioIcon();
      audio.title = lastName;
    }

    /* LOGO 按钮：播放 / 暂停 */
    window.toggleAudio = function () {
      const a = document.getElementById('audio-element');
      if (!a) return;
      if (a.paused) { const p = a.play(); if (p && p.catch) p.catch(()=>{}); }
      else { a.pause(); }
      syncAudioIcon();
    };
    function syncAudioIcon() {
      const a = document.getElementById('audio-element');
      const btn = document.getElementById('logoBtn');
      const note = document.getElementById('logoNote');
      const playing = !!a && !a.paused && !a.ended;
      if (note) note.textContent = playing ? '❚❚' : '♪';
      if (btn) btn.classList.toggle('playing', playing);
    }

      processDialogue();
