/* ==================== SillyTavern 酒馆消息发送 ====================
 * 主页.html 三级降级策略：
 *   1. sendUserMessage(msg)  —— 酒馆注入的全局函数（首选）
 *   2. DOM 直插法：查找酒馆 #send_textarea + #send_but，直接 set value + click（备选）
 *   3. navigator.clipboard.writeText  —— 复制到剪贴板（再降级）
 *   4. alert —— 最终降级提示
 * 返回 true 表示已成功送出，false 表示降级提示
 */
function sendToTavern(msg){
  if(!msg||typeof msg!=='string')return false;

  /* ① 优先：酒馆注入的全局函数 */
  try{
    if(typeof sendUserMessage==='function'){sendUserMessage(msg);return true;}
  }catch(e){__dbg('sendToTavern.sendUserMessage.err',{msg:String(e).slice(0,120)});}

  /* ② DOM 直插法：完美家庭主页，操作酒馆页面的 textarea + 发送按钮 */
  var tryDoc=function(doc){
    try{
      var inp=doc.querySelector('#send_textarea')||
              doc.querySelector('textarea[placeholder]:not(#playerInput)');
      var btn=doc.querySelector('#send_but')||
              doc.querySelector('#send_but_sheld')||
              doc.querySelector('button[title*="Send"]')||
              doc.querySelector('button[aria-label*="发送"]:not(#castBtn)')||
              doc.querySelector('button[aria-label*="Send"]');
      if(inp&&btn){
        inp.value=msg;
        inp.dispatchEvent(new Event('input',{bubbles:true}));
        inp.dispatchEvent(new Event('change',{bubbles:true}));
        setTimeout(function(){btn.click();},80);
        return true;
      }
    }catch(e){__dbg('sendToTavern.tryDoc.err',{msg:String(e).slice(0,120)});}
    return false;
  };
  try{
    if(tryDoc(document))return true;
    if(window.parent&&window.parent!==window){
      if(tryDoc(window.parent.document))return true;
    }
  }catch(e){__dbg('sendToTavern.parent.err',{msg:String(e).slice(0,120)});}

  /* ③ 降级：复制到剪贴板 */
  try{
    if(navigator.clipboard&&navigator.clipboard.writeText){
      navigator.clipboard.writeText(msg).then(function(){
        showToast('已复制到剪贴板，请粘贴发送','ri-clipboard-fill');
      },function(){
        alert('剧情指令已生成（请手动粘贴到酒馆）:\n\n'+msg);
      });
      return true;
    }
  }catch(e){}
  alert('剧情指令已生成（请手动粘贴到酒馆）:\n\n'+msg);
  return false;
}

/* 获取某角色某场景的最大编号，不存在返回0
 * 支持 NFC 归一化 + 去空格模糊匹配（角色名 & 场景名均支持） */
function cgGetMaxCount(cat,char,scene){
  var catData=CG_IMAGE_LISTS[cat];
  if(!catData)return 0;
  char=(char||'').trim();scene=(scene||'').trim();
  if(!char||!scene)return 0;
  /* ① 角色匹配 */
  var charData=catData[char];
  if(!charData){
    var normC=char.normalize('NFC');
    for(var ck in catData){
      if(ck.trim().normalize('NFC')===normC||ck.replace(/\s/g,'')===char.replace(/\s/g,'')){charData=catData[ck];break;}
    }
  }
  if(!charData)return 0;
  /* ② 场景匹配 */
  if(charData[scene])return charData[scene];
  var normS=scene.normalize('NFC');
  for(var sk in charData){
    if(sk.trim().normalize('NFC')===normS||sk.replace(/\s/g,'')===scene.replace(/\s/g,''))return charData[sk];
  }
  return 0;
}

function _cgCharExists(char){
  if(CG_CHAR_DOMAIN[char])return true;
  var nc=char.trim().normalize('NFC');
  for(var c in CG_CHAR_DOMAIN){if(c.trim().normalize('NFC')===nc||c.replace(/\s/g,'')===char.replace(/\s/g,''))return true;}
  return false;
}
function _cgPickSubChar(origChar){
  var pool=CG_CHARACTERS.filter(function(c){return c!==origChar;});
  return pool.length?pool[Math.floor(Math.random()*pool.length)]:CG_CHARACTERS[0];
}
function _cgPickSubType(cat,excludeType){
  var src=cat==='SFW'?_SFW_COMMON:_NSFW_COMMON;
  var keys=Object.keys(src).filter(function(k){return k!==excludeType;});
  if(!keys.length)keys=Object.keys(src);
  return keys.length?keys[Math.floor(Math.random()*keys.length)]:'';
}
function _cgSceneExists(cat,scene){
  var src=cat==='SFW'?_SFW_COMMON:_NSFW_COMMON;
  if(src[scene])return true;
  var ns=scene.trim().normalize('NFC');
  for(var k in src){if(k.trim().normalize('NFC')===ns||k.replace(/\s/g,'')===scene.replace(/\s/g,''))return true;}
  return false;
}

/* ════════ CG 图片 IndexedDB 缓存 ════════
 * 将已查看的 CG .webp 图片 blob 存入 IndexedDB，
 * 下次打开画廊直接从本地读取，无需重新下载。
 * - DB名: sjjy_cg_imgdb / Store名: images / Key: 远程URL
 * - Value: { blob: Blob, ts: number }
 * - 容量上限: 200MB，超限自动淘汰最旧条目
 */
var CG_IMG_DB=null;
var CG_IMG_DB_NAME='sjjy_cg_imgdb';
var CG_IMG_STORE='images';
var CG_IMG_MAX_SIZE=200*1024*1024; /* 200MB上限 */
var _cgBlobUrls={}; /* url→blobUrl 映射，用于追踪和释放 */

/* 打开/初始化 IndexedDB */
function cgImgDbOpen(){
  if(CG_IMG_DB)return Promise.resolve(CG_IMG_DB);
  return new Promise(function(ok,fail){
    try{
      var req=indexedDB.open(CG_IMG_DB_NAME,1);
      req.onupgradeneeded=function(e){
        var db=e.target.result;
        if(!db.objectStoreNames.contains(CG_IMG_STORE)){
          db.createObjectStore(CG_IMG_STORE,{keyPath:'url'});
        }
      };
      req.onsuccess=function(e){
        CG_IMG_DB=e.target.result;
        ok(CG_IMG_DB);
      };
      req.onerror=function(e){fail(e.target.error);};
    }catch(e){fail(e);}
  });
}

/* 从 IndexedDB 读取一张缓存图片，返回 blobUrl 或 null */
function cgImgGet(remoteUrl){
  return cgImgDbOpen().then(function(db){
    return new Promise(function(ok){
      try{
        var tx=db.transaction(CG_IMG_STORE,'readonly');
        var store=tx.objectStore(CG_IMG_STORE);
        var req=store.get(remoteUrl);
        req.onsuccess=function(e){
          var rec=e.target.result;
          if(rec&&rec.blob){
            /* 释放旧的 blobUrl（如有） */
            if(_cgBlobUrls[remoteUrl])URL.revokeObjectURL(_cgBlobUrls[remoteUrl]);
            var bUrl=URL.createObjectURL(rec.blob);
            _cgBlobUrls[remoteUrl]=bUrl;
            ok(bUrl);
          }else{ok(null);}
        };
        req.onerror=function(){ok(null);};
      }catch(e){ok(null);}
    });
  });
}

/* 将一张图片写入 IndexedDB 缓存 */
function cgImgPut(remoteUrl,blob){
  return cgImgDbOpen().then(function(db){
    return new Promise(function(ok,fail){
      try{
        var tx=db.transaction(CG_IMG_STORE,'readwrite');
        var store=tx.objectStore(CG_IMG_STORE);
        store.put({url:remoteUrl,blob:blob,ts:Date.now()});
        tx.oncomplete=function(){ok();};
        tx.onerror=function(e){fail(e.target.error);};
      }catch(e){fail(e);}
    });
  });
}

/* 淘汰旧缓存直到总容量低于上限 */
function cgImgEvict(){
  return cgImgDbOpen().then(function(db){
    return new Promise(function(ok){
      try{
        var tx=db.transaction(CG_IMG_STORE,'readonly');
        var store=tx.objectStore(CG_IMG_STORE);
        var req=store.openCursor();
        var entries=[];
        req.onsuccess=function(e){
          var cursor=e.target.result;
          if(cursor){
            var rec=cursor.value;
            entries.push({url:rec.url,size:rec.blob?rec.blob.size:0,ts:rec.ts||0});
            cursor.continue();
          }else{
            /* 统计总容量 */
            var total=0;
            entries.forEach(function(en){total+=en.size;});
            if(total<CG_IMG_MAX_SIZE){ok();return;}
            /* 按时间排序，最旧优先淘汰 */
            entries.sort(function(a,b){return a.ts-b.ts;});
            var delTx=db.transaction(CG_IMG_STORE,'readwrite');
            var delStore=delTx.objectStore(CG_IMG_STORE);
            var freed=0;
            for(var i=0;i<entries.length&&freed<(total-CG_IMG_MAX_SIZE*0.8);i++){
              delStore.delete(entries[i].url);
              freed+=entries[i].size;
            }
            delTx.oncomplete=function(){ok();};
            delTx.onerror=function(){ok();};
          }
        };
        req.onerror=function(){ok();};
      }catch(e){ok();}
    });
  });
}

/* 获取缓存统计信息：{ count: Number, size: Number(字节) } */
function cgImgStats(){
  return cgImgDbOpen().then(function(db){
    return new Promise(function(ok){
      try{
        var tx=db.transaction(CG_IMG_STORE,'readonly');
        var store=tx.objectStore(CG_IMG_STORE);
        var req=store.openCursor();
        var count=0,size=0;
        req.onsuccess=function(e){
          var cursor=e.target.result;
          if(cursor){
            count++;
            var b=cursor.value.blob;
            if(b)size+=b.size;
            cursor.continue();
          }else{
            ok({count:count,size:size});
          }
        };
        req.onerror=function(){ok({count:0,size:0});};
      }catch(e){ok({count:0,size:0});}
    });
  });
}

/* 清空所有图片缓存 */
function cgImgClear(){
  /* 释放所有 blobUrl */
  for(var u in _cgBlobUrls){URL.revokeObjectURL(_cgBlobUrls[u]);}
  _cgBlobUrls={};
  return cgImgDbOpen().then(function(db){
    return new Promise(function(ok){
      try{
        var tx=db.transaction(CG_IMG_STORE,'readwrite');
        tx.objectStore(CG_IMG_STORE).clear();
        tx.oncomplete=function(){ok(true);};
        tx.onerror=function(){ok(false);};
      }catch(e){ok(false);}
    });
  });
}

/* ════════ CG 图片缓存核心：缓存优先加载 ════════
 * 1. 查 IndexedDB → 有 → 直接返回 blobUrl
 * 2. 无 → fetch 远程 → 存 IndexedDB → 返回 blobUrl
 * 3. fetch 失败 → 回退远程URL
 */
function cgLoadImg(remoteUrl){
  return cgImgGet(remoteUrl).then(function(bUrl){
    if(bUrl)return bUrl;
    /* 未缓存，fetch 远程 */
    return fetch(remoteUrl,{mode:'cors',credentials:'omit'}).then(function(r){
      if(!r.ok)throw new Error('HTTP '+r.status);
      return r.blob();
    }).then(function(blob){
      /* 存入缓存（异步，不阻塞显示） */
      cgImgPut(remoteUrl,blob).then(function(){
        /* 存完后淘汰检查 */
        cgImgEvict();
      });
      /* 释放旧 blobUrl（如有） */
      if(_cgBlobUrls[remoteUrl])URL.revokeObjectURL(_cgBlobUrls[remoteUrl]);
      var newUrl=URL.createObjectURL(blob);
      _cgBlobUrls[remoteUrl]=newUrl;
      return newUrl;
    }).catch(function(){
      /* fetch 失败，回退远程URL */
      return remoteUrl;
    });
  }).catch(function(){
    /* IndexedDB 不可用，回退远程URL */
    return remoteUrl;
  });
}

/* 记录一条 CG 解锁（去重，需通过 IMAGE_LISTS 验证） */
function cgRecord(cat,char,scene){
  if(cat!=='SFW'&&cat!=='NSFW')return;
  char=(char||'').trim();scene=(scene||'').trim();
  if(!char||!scene)return;
  /* 验证角色和场景是否都在 IMAGE_LISTS 中存在 */
  if(!cgGetMaxCount(cat,char,scene))return;
  /* NSFW 视频需好感度 100 才可解锁（扫描解锁同样受限；一键解锁按钮亦仅在好感度满时出现） */
  if(cat==='NSFW'&&!cgCharFavorMax(char))return;
  var c=cgGetCache();
  if(!c[cat])c[cat]={};
  if(!c[cat][char])c[cat][char]=[];
  if(c[cat][char].indexOf(scene)<0){
    c[cat][char].push(scene);
    cgSaveCache(c);
  }
}

/* 扫描酒馆聊天记录，自动收录 CG
 * 识别两种格式：
 *   1. 文本标签：<img>SFW/角色名/插画名</img> 或 <img>NSFW/角色名/插画名{{roll: d3}}</img>
 *   2. 已渲染的 <img src="https://zyxjack123.top/足控天堂/SFW/角色名/插画名1.webp">
 * 所有角色共用 IMAGE_LISTS，只收录配置中存在的合法场景。
 */
/* ── 内部：从标签内容（如 "SFW/莱昂纳多·达·芬奇/生气1" 或 "NSFW/角色/场景{{roll: d3}}"）解析并记录CG ──
 * 替换优先级：
 *   1. 角色不存在 + 场景存在 → 只换角色，保留场景
 *   2. 角色不存在 + 场景也不存在 → 换角色 + 换场景
 *   3. 角色存在 + 场景不存在 → 只换场景 */
function cgParseAndRecord(content){
  content=(content||'').trim();
  if(!content)return;
  var parts=content.split('/');
  if(parts.length<3)return;
  var cat=parts[0].trim();
  if(cat!=='SFW'&&cat!=='NSFW')return;
  var char=cgSafeDecode(parts[1].trim());
  var fileName=cgSafeDecode(parts.slice(2).join('/').trim());
  fileName=fileName.replace(/\{\{roll:\s*d\d+\}\}/i,'').trim();
  /* 去掉末尾数字得到场景名 */
  var sm=fileName.match(/^(.+?)(\d+)$/);
  var scene=sm?sm[1]:fileName;
  var charOk=_cgCharExists(char);
  var sceneOk=_cgSceneExists(cat,scene);
  /* 只有角色和场景都存在时才记录解锁，不进行随机替换 */
  if(charOk&&sceneOk){
    cgRecord(cat,char,scene);
  }
}

function cgScanChat(){
  try{
    /* ═══ 方案①：直接读取 SillyTavern 原始 chat 数组（最佳，未被浏览器解析污染） ═══ */
    var targetWindow=window;

    if(targetWindow.SillyTavern&&targetWindow.SillyTavern.getContext){
      var ctx=targetWindow.SillyTavern.getContext();
      var chat=ctx.chat||[];
      /* 简单正则：匹配 <img>任意内容</img>（参考 loveli 状态栏机制） */
      var tagPtn=/<img>([^<]+)<\/img>/gi;
      /* 转义版正则：匹配 &lt;img&gt;任意内容&lt;/img&gt; */
      var escPtn=/&lt;img&gt;([^<]+)&lt;\/img&gt;/gi;
      for(var i=0;i<chat.length;i++){
        var mes=chat[i]&&(chat[i].mes||'');
        if(!mes)continue;
        var m;
        tagPtn.lastIndex=0;
        while((m=tagPtn.exec(mes))!==null) cgParseAndRecord(m[1]);
        escPtn.lastIndex=0;
        while((m=escPtn.exec(mes))!==null) cgParseAndRecord(m[1]);
      }
    }

    /* ═══ 方案②：扫描 DOM 作为兜底（适用于 SillyTavern API 不可用时） ═══ */
    var docs=[document];
    try{if(window.parent&&window.parent.document&&window.parent.document!==document)docs.push(window.parent.document);}catch(e){}
    /* DOM 用到的正则 */
    var tagPtn2=/<img>([^<]+)<\/img>/gi;
    var escPtn2=/&lt;img&gt;([^<]+)&lt;\/img&gt;/gi;
    /* 裸路径正则：浏览器把<img>当void元素解析后，textContent只剩 SFW/角色/场景N */
    var barePtn=/\b(SFW|NSFW)\/([^\/\s<&,;]+)\/([^\s<&,;\])}]+)/gi;

    docs.forEach(function(doc){
      var mesEls=doc.querySelectorAll('.mes_text, .mes .text, [class*="message-text"]');
      mesEls.forEach(function(el){
        var m;
        /* a) textContent 用完整标签正则 */
        var text=el.textContent||'';
        tagPtn2.lastIndex=0;
        while((m=tagPtn2.exec(text))!==null) cgParseAndRecord(m[1]);
        /* b) textContent 用转义标签正则 */
        escPtn2.lastIndex=0;
        while((m=escPtn2.exec(text))!==null) cgParseAndRecord(m[1]);
        /* c) textContent 用裸路径正则（浏览器void元素场景） */
        barePtn.lastIndex=0;
        while((m=barePtn.exec(text))!==null){
          cgParseAndRecord(m[1]+'/'+m[2]+'/'+m[3]);
        }
        /* d) innerHTML 用转义标签正则 */
        var html=el.innerHTML||'';
        escPtn2.lastIndex=0;
        while((m=escPtn2.exec(html))!==null) cgParseAndRecord(m[1]);
      });
      /* ② 扫描已渲染的实际 <img> 元素 */
      var imgs=doc.querySelectorAll('img[src*="zyxjack123.top/足控天堂/"]');
      imgs.forEach(function(img){
        var parts=img.src.split('/');
        var idx=parts.findIndex(function(p){return p==='SFW'||p==='NSFW';});
        if(idx<0||idx+2>=parts.length)return;
        var cat=parts[idx];
        var char=cgSafeDecode(parts[idx+1]);
        var fn=cgSafeDecode(parts[idx+2]).replace(/\.webp$/,'');
        var sm=fn.match(/^(.+?)(\d+)$/);
        var scene=sm?sm[1]:fn;
        cgRecord(cat,char,scene);
      });
    });
  }catch(e){console.error('CG扫描失败:',e);}
}

 * 对每个 remoteUrl 调 cgLoadImg → 替换 background-image
 */
function cgApplyImgCache(container){
  if(!container)container=document;
  var els=container.querySelectorAll('[data-cg-remote]');
  if(!els.length)return;
  els.forEach(function(el){
    var remoteUrl=el.getAttribute('data-cg-remote');
    if(!remoteUrl)return;
    el.removeAttribute('data-cg-remote'); /* 防止重复处理 */
    cgLoadImg(remoteUrl).then(function(bUrl){
      if(bUrl&&bUrl!==remoteUrl){
        el.style.backgroundImage='url(\''+bUrl+'\')';
      }
    });
  });
}

// ==================== 核心过滤函数 ====================
// 过滤 $template（原搬自足控天堂状态栏美化.html）
function filterTemplates(obj) {
    if (!obj || typeof obj !== 'object') return obj;

    if (Array.isArray(obj)) return obj;

    const filtered = {};
    for (const key in obj) {
        if (key === '$template') continue;
        if (typeof obj[key] === 'object' && obj[key] !== null) {
            filtered[key] = filterTemplates(obj[key]);
        } else {
            filtered[key] = obj[key];
        }
    }
    return filtered;
}

// ==================== ERA 工具函数（data-era 驱动） ====================
// 通用取值函数（解决 data-era 路径，从 currentStat 取值）
function eraGet(path, defaultVal = '') {
    if (!currentStat) return defaultVal;
    const parts = path.split('.');
    let val = currentStat;
    for (const p of parts) {
        if (val == null) return defaultVal;
        val = val[p];
    }
    return val !== undefined && val !== null ? val : defaultVal;
}

function eraGetDef(path, defaultVal) {
    return eraGet(path, defaultVal);
}

// 判断对象是否为模板键（$ 开头）
function isTemplate(key){return key&&key.charAt(0)==='$'}

// 获取对象下的实际数据（过滤 $template）
function getItems(obj) {
    if (!obj || typeof obj !== 'object') return [];
    const clean = filterTemplates(obj);

// 渲染指定索引的附近角色到对话栏
function renderNearbyAt(idx) {
    if (!__nearbyRotateItems.length) return;
    if (idx >= __nearbyRotateItems.length) idx = 0;
    __nearbyIdx = idx;
    const ch = __nearbyRotateItems[idx];

    const nameEl = document.getElementById('speakerName');
    const roleEl = document.querySelector('[data-era-nearby="role"]');
    const thoughtEl = document.getElementById('dialogText');
    if (nameEl) nameEl.textContent = ch._key || '主播';
    if (roleEl) {
        const num = ch.数值 || {};
        const roleParts = [
            ch.足控依赖阶段 || '',
            ch.是否签约 === true ? '已签约' : (ch.是否签约 === false ? '未签约' : ''),
            ch.是否在场 === true ? '在场' : '',
            num.足控依赖度 != null ? '依赖' + num.足控依赖度 : '',
            num.好感度 != null ? '好感' + num.好感度 : ''
        ].filter(Boolean);
        roleEl.textContent = roleParts.join(' · ') || '——';
    }
    if (thoughtEl) {
        const thought = ch.内心想法 || '……';
        thoughtEl.innerHTML = thought + '<span class="cursor"></span>';
    }

    // 头像
    const glyph = document.getElementById('speakerGlyph');
    if (glyph) {
        const img = (typeof CHAR_NAME_AV !== 'undefined' && CHAR_NAME_AV[ch._key])
            || (typeof charImgMap !== 'undefined' && charImgMap[ch._key])
        if (img) {
            glyph.classList.add('has-img');
            glyph.innerHTML = '';
            const imgEl = document.createElement('img');
            imgEl.src = img;
            imgEl.alt = ch._key || '主播';
            imgEl.onerror = function(){
                this.style.display = 'none';
                this.parentNode.classList.remove('has-img');
                this.parentNode.innerHTML = '<i class="ri-user-3-fill"></i>';
            };
            glyph.appendChild(imgEl);
        } else {
            glyph.classList.remove('has-img');
            glyph.innerHTML = '<i class="ri-user-3-fill"></i>';
        }
    }
}

function startNearbyCarousel() {
    if (__nearbyTimer) { clearInterval(__nearbyTimer); __nearbyTimer = null; }
    if (!__nearbyRotateItems.length) return;
    __nearbyIdx = 0;
    renderNearbyAt(__nearbyIdx);
    if (__nearbyRotateItems.length > 1) {
        __nearbyTimer = setInterval(function(){
            if (document.hidden) return;   /* 后台不轮播，省电 */
            __nearbyIdx = (__nearbyIdx + 1) % __nearbyRotateItems.length;
            renderNearbyAt(__nearbyIdx);
        }, __NEARBY_INTERVAL);
    }
}

// ERA 数据更新时调用：刷新附近角色列表，轮播内心想法最新变更的角色
function updateNearbyCharacter() {
    const archive = eraGet('主播档案') || {};
    __nearbyItems = getItems(archive);
    const nameEl = document.getElementById('speakerName');
    const roleEl = document.querySelector('[data-era-nearby="role"]');
    const thoughtEl = document.getElementById('dialogText');

    if (!__nearbyItems.length) {
        if (__nearbyTimer) { clearInterval(__nearbyTimer); __nearbyTimer = null; }
        if (nameEl) nameEl.textContent = '当前主播';
        if (roleEl) roleEl.textContent = '——';
        if (thoughtEl) thoughtEl.innerHTML = '暂无主播的内心想法。<span class="cursor"></span>';
        __nearbyRotateItems = [];
        /* 不清空 __prevThoughts，保留历史记录避免下次全员误判为"新" */
        return;
    }

    /* ① 检测内心想法变更（已有记录且文本不同且新想法非空 → 真正变更） */
    var changedItems = [];
    __nearbyItems.forEach(function(ch){
        var name = ch._key || '';
        var thought = ch.内心想法 || '';
        var prev = __prevThoughts[name];
        /* 只有新想法非空、且与上次不同才算变更 */
        if(prev !== undefined && thought && prev !== thought){
            changedItems.push(ch);
        }
        __prevThoughts[name] = thought;
    });

    /* ② 有变更 → 只轮播变更的角色 */
    if(changedItems.length > 0){
        __nearbyRotateItems = changedItems;
        startNearbyCarousel();
        return;
    }

    /* ③ 无变更 → 首次加载初始化（只包含有内心想法的角色），之后保持上次轮播不变 */
    if(__nearbyRotateItems.length === 0){
        __nearbyRotateItems = __nearbyItems.filter(function(ch){
            return ch.内心想法 && ch.内心想法.trim();
        });
        if(!__nearbyRotateItems.length){
            __nearbyRotateItems = __nearbyItems.slice();
        }
        startNearbyCarousel();
    }
    /* 否则保持现有轮播继续，不重启 */
}

function setupERAListeners() {
    // 确保 eventOn / eventEmit 存在（酒馆助手环境）
    if (typeof eventOn === 'undefined') {
        console.warn("⚠️ eventOn 未定义，请确认在酒馆助手环境中运行");
        return;
    }

    eventOn('era:writeDone', (detail) => {
        console.log('%c[ERA] writeDone', 'color:#67e8f9', detail);
        if (detail.statWithoutMeta) {
            renderAll(detail.statWithoutMeta);
        }
        /* CG画廊：ERA写入后扫描聊天记录解锁新CG */
        setTimeout(function(){cgScanChat();},300);
    });

    eventOn('era:queryResult', (detail) => {
        if (detail.result && detail.result.error) {
            console.error('ERA 查询失败:', detail.result.error);
            return;
        }
        if (detail.queryType === 'getCurrentVars' && detail.result) {
            renderAll(detail.result.statWithoutMeta || detail.result.stat);
        }
    });

    // 初始化获取一次数据
    setTimeout(() => {
        if (typeof eventEmit === 'function') {
            eventEmit('era:getCurrentVars');
        }
    }, 1200);
}
