function initVideo(){
  var vid = document.getElementById('cover-vid');
  if(!vid) return;
  vid.muted = false;
  var p = vid.play();
  if(p && p.catch) p.catch(function(){
    vid.muted = true;
    vid.play().catch(function(){});
  });
  function unmute(){
    vid.muted = false;
    vid.play().catch(function(){});
    document.removeEventListener('click', unmute);
    document.removeEventListener('keydown', unmute);
  }
  document.addEventListener('click', unmute);
  document.addEventListener('keydown', unmute);
}

/* 填入 SillyTavern 底部消息输入框 #send_textarea（参考「碎界纪元主页」机制）
 * 仅填入、不调用任何发送 API；用户可检查后手动发送。
 * SillyTavern 为受控组件，必须用原生 setter + input/change 事件触发框架状态更新；
 * 已内嵌 iframe 时同时搜索父文档（SillyTavern 主窗口）。 */
function fillSendTextarea(fullText){
  function setNativeValue(el, val){
    try{
      var proto = el instanceof HTMLInputElement ? HTMLInputElement.prototype : HTMLTextAreaElement.prototype;
      var setter = Object.getOwnPropertyDescriptor(proto, 'value').set;
      setter.call(el, val);
    }catch(e){ el.value = val; }
  }
  function triggerFullEvents(el){
    var evtTypes = ['input','change'];
    for(var i=0;i<evtTypes.length;i++){
      el.dispatchEvent(new Event(evtTypes[i], {bubbles:true, cancelable:true}));
      el.dispatchEvent(new InputEvent('input', {bubbles:true, cancelable:true, data:el.value, inputType:'insertText'}));
    }
    if(typeof $ !== 'undefined' && $(el).length){ $(el).trigger('input').trigger('change'); }
  }
  var candidates = [
    function(){ return document.querySelector('#send_textarea'); },
    function(){ return document.querySelector('textarea#send_textarea'); },
    function(){ return document.querySelector('textarea[placeholder*="输入"]'); },
    function(){ return document.querySelector('textarea[placeholder*="发送"]'); },
    function(){ return document.querySelector('textarea.mdHotkeys'); }
  ];
  var parentDocs = [];
  try{
    if(window.parent && window.parent !== window) parentDocs.push(window.parent.document);
    if(window.opener) parentDocs.push(window.opener.document);
    if(window.parent && window.parent.parent && window.parent.parent !== window.parent) parentDocs.push(window.parent.parent.document);
  }catch(e){}
  for(var pi=0; pi<parentDocs.length; pi++){
    (function(pd){
      candidates.push(
        function(){ return pd && pd.querySelector('#send_textarea'); },
        function(){ return pd && pd.querySelector('textarea#send_textarea'); },
        function(){ return pd && pd.querySelector('textarea[placeholder]'); },
        function(){ return pd && pd.querySelector('textarea.mdHotkeys'); }
      );
    })(parentDocs[pi]);
  }
  for(var ci=0; ci<candidates.length; ci++){
    try{
      var inp = candidates[ci]();
      if(!inp || !inp.tagName) continue;
      setNativeValue(inp, fullText);
      triggerFullEvents(inp);
      inp.focus();
      if(inp.setSelectionRange) inp.setSelectionRange(fullText.length, fullText.length);
      console.log('[足控天堂] 成功填入 #send_textarea, 长度:', fullText.length);
      return true;
    }catch(e){ console.warn('[足控天堂] 候选失败:', e.message); continue; }
  }
  return false;
}
function launch(){
  var msg = buildPrompt();
  if(fillSendTextarea(msg)){ tip('✦ 已填入消息输入框，请检查后手动发送'); return; }
  copyPrompt();
}
function copyPrompt(){
  var msg = buildPrompt();
  if(navigator.clipboard && navigator.clipboard.writeText){
    navigator.clipboard.writeText(msg).then(function(){ tip('已复制到剪贴板。'); }, function(){ fallbackCopy(msg); });
  } else { fallbackCopy(msg); }
}
function fallbackCopy(msg){
  var ta = document.createElement('textarea');
  ta.value = msg; document.body.appendChild(ta); ta.select();
  try{ document.execCommand('copy'); tip('已复制到剪贴板。'); }
  catch(e){ alert(msg); }
  document.body.removeChild(ta);
}
